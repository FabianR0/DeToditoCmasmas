#include <iostream>
#include <vector>
using namespace std;

void print (vector<int> v){
    for(int i = 0; i  <v.size(); i++ ){
        cout << " " << v.at(i);
    }
    cout<<endl;
}

int main()
{
   vector<int> digits;
     for(int i = 0; i < 10;i++ ){
        digits.push_back(i);
    }
    
    vector<vector<int>>x;
     for(int i = 0; i < 10;i++ ){
        x.push_back(digits);
    }
     x[3].pop_back();
    for(int i =0; i<10;i++ ){
        print(x.at(i));
    }
   
    print(digits);
    
    return 0;
}
