#include <iostream>

#include <stdlib.h>
using namespace std;
bool Sudoku_R1(vector<vector<int>>y){
    cout<<"--1 region--"<<endl;  
     for (int i = 0; i < 1; ++i) {
        for (int j = 0; j < 3; ++j) {
            for (int k = 0; k < 3; ++k) {
                cout << y[j][k]<< " ";
                if (y[j][k]==y[j][k+1]) {
                    return false;
                }
            }cout << endl;
        }
    }
    return true;
}

bool Sudoku_G(vector<vector<int>>w){
    
     for (int i = 0; i < 1; ++i) {
        for (int j = 0; j < w.size(); ++j) {
            for (int k = 0; k < w[j].size(); ++k) {
                cout  << w[j][k]<<  " ";
                if (w[j][k]==w[j][k+1]) {
                    return false;
                }
            }cout << endl;
        }
    }
    return true;
}
