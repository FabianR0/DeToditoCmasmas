int main()
{
   vector<vector<int>>x{
       {1,4,5,6,9,7,8,2,3},
       {8,6,7,1,2,3,5,4,9},
       {9,3,2,5,8,4,1,7,6},

       {6,2,4,3,5,9,7,1,8},
       {7,1,3,2,6,8,4,9,5},
       {5,8,9,4,7,1,6,3,2}};
    

	 for(int r=0; r<=8; r++){
    for(int i=0; i <=8; i++){
        for(int j = i+1;j <=8; j++){
            if (x[r][i]==x[r][j])
            exit( 0 );
            }
        }
    }
    cout << " el sudoku esta bien hecho  "<< endl;

    cout <<x.size()<< endl;
    cout <<x[0].size()<< endl;
    cout <<x[2][1]<< endl;
