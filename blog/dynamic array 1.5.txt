bool Sudoku_R(vector<vector<int>>y){
    cout<<"------1 region--------"<<endl;  
     for (int i = 0; i < 1; ++i) {
        for (int j = 0; j < 3; ++j) {
            for (int k = 0; k < 3; ++k) {
                cout << "Element at x[" << i << "][" << j << "][" << k << "] = " << y[j][k]<< endl;
                if (y[j][k]==y[j][k+1]) {
                    return false;
                }
            }
        }
    }
    return true;
}
